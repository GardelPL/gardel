# Parser cases for GHTTP_TEST, read on standard input:
#
#   build/ghttp_test_bin.exe < ghttp/tests/cases.txt
#
# Each line is a request, ` => `, and what the parser must say of it, whole and cut into
# two reads at every byte. In the request, \r, \n, \t and \\ stand for CR, LF, tab and a
# backslash. Lines that start with # and empty lines are skipped.

# Accepted.
GET /any/path HTTP/1.1\r\n\r\n => GET /any/path HTTP/1.1 headers 0 body [] rest []
GET /api/v1 HTTP/1.1\r\nHost: x\r\n\r\n => GET /api/v1 HTTP/1.1 headers 1 body [] rest []
GET /nope HTTP/1.1\r\nHost: x\r\n\r\n => GET /nope HTTP/1.1 headers 1 body [] rest []
POST /users HTTP/1.1\r\nHost: x\r\nContent-Length: 2\r\n\r\nhi => POST /users HTTP/1.1 headers 2 body [hi] rest []
PUT /a?b=1 HTTP/1.0\r\ncontent-length:  5 \r\nX-A:\tb\r\n\r\nhello => PUT /a?b=1 HTTP/1.0 headers 2 body [hello] rest []
POST /x HTTP/1.1\r\nContent-Length: 2\r\nContent-Length: 2\r\n\r\nok => POST /x HTTP/1.1 headers 2 body [ok] rest []
DELETE /users/42 HTTP/1.1\r\n\r\n => DELETE /users/42 HTTP/1.1 headers 0 body [] rest []

# Field names ignore case: each spelling finds `Content-Type`, shown as `type [...]`.
GET /t HTTP/1.1\r\nContent-Type: text/plain\r\n\r\n => GET /t HTTP/1.1 headers 1 body [] rest [] type [text/plain]
GET /t HTTP/1.1\r\ncontent-type: a/b\r\n\r\n => GET /t HTTP/1.1 headers 1 body [] rest [] type [a/b]
GET /t HTTP/1.1\r\nCONTENT-TYPE:x\r\nHost: h\r\n\r\n => GET /t HTTP/1.1 headers 2 body [] rest [] type [x]
GET /t HTTP/1.1\r\nContent-Typo: x\r\n\r\n => GET /t HTTP/1.1 headers 1 body [] rest []

# What follows a request is handed back, not lost.
GET /a HTTP/1.1\r\n\r\nGET /b => GET /a HTTP/1.1 headers 0 body [] rest [GET /b]
POST /p HTTP/1.1\r\nContent-Length: 3\r\n\r\nabcGET => POST /p HTTP/1.1 headers 1 body [abc] rest [GET]

POST /p HTTP/1.1\r\nContent-Length: 4\r\n\r\nab => more 41

# The request writer (F.8). `write <request> => <bytes>`: the request is parsed, its own
# framing fields are dropped, and it must be written as exactly these bytes, which must
# parse back to it. Every request accepted above is also written and parsed back.
write GET /a HTTP/1.1\r\nHost: x\r\nAccept: */*\r\n\r\n => GET /a HTTP/1.1\r\nHost: x\r\nAccept: */*\r\n\r\n
write POST /users HTTP/1.1\r\nHost: x\r\nContent-Length: 2\r\n\r\nhi => POST /users HTTP/1.1\r\nHost: x\r\nContent-Length: 2\r\n\r\nhi
write POST /p HTTP/1.1\r\n\r\n => POST /p HTTP/1.1\r\nContent-Length: 0\r\n\r\n
write PUT /a?b=1 HTTP/1.0\r\ncontent-length:  5 \r\nX-A:\tb\r\n\r\nhello => PUT /a?b=1 HTTP/1.0\r\nX-A: b\r\nContent-Length: 5\r\n\r\nhello
write DELETE /users/42 HTTP/1.1\r\n\r\n => DELETE /users/42 HTTP/1.1\r\n\r\n

# The response reader (F.8). `response <method> <bytes> => <what>` reads the answer to a
# request with that method; `closed` lines then close the connection. Each is read whole and
# cut into two reads at every byte. A response shows as `<code> <version> headers <n> body
# [...] rest [...]`.
response GET HTTP/1.1 200 OK\r\nContent-Length: 2\r\n\r\nhi => 200 HTTP/1.1 headers 1 body [hi] rest []
response GET HTTP/1.1 200\r\nContent-Length: 0\r\n\r\n => 200 HTTP/1.1 headers 1 body [] rest []
response GET HTTP/1.1 201 Created\r\nContent-Type: application/json\r\nContent-Length: 7\r\n\r\n{"a":1} => 201 HTTP/1.1 headers 2 body [{"a":1}] rest []
response GET HTTP/1.0 404 Not Found\r\nContent-Length: 3\r\n\r\nno!extra => 404 HTTP/1.0 headers 1 body [no!] rest [extra]
response GET HTTP/1.1 200 OK\r\ncontent-length:  4 \r\n\r\nabcd => 200 HTTP/1.1 headers 1 body [abcd] rest []
